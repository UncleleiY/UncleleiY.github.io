VBA脚本：用于数据清洗

辅助脚本（by Uncle-yang）

借助AI写的一个小脚本，用于自动筛选异常类型并分类：

' ================================================================
' 主程序：添加分类列并进行数据分类处理
' 功能：根据异常说明内容自动为数据行添加分类标记
' ================================================================
Sub AddClassificationColumn()
    ' 声明变量
    Dim ws As Worksheet          ' 工作表对象
    Dim lastCol As Long, lastRow As Long                 ' 最后列和最后行号
    Dim i As Long, j As Long                            ' 循环计数器
    Dim classificationCol As Long, exceptionCol As Long, stepNameCol As Long, handlingCol As Long ' 列位置
    Dim exceptionText As String, stepNameText As String, handlingText As String ' 文本内容
    
    ' 设置工作表对象 - 使用当前活动工作表
    Set ws = ThisWorkbook.ActiveSheet
    
    ' 查找关键列位置（在表头行中查找指定列名）
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For i = 1 To lastCol
        Select Case ws.Cells(1, i).Value
            Case "分类": classificationCol = i           ' 分类列位置
            Case "异常说明": exceptionCol = i           ' 异常说明列位置
            Case "步骤名称": stepNameCol = i            ' 步骤名称列位置
            Case "处置说明": handling = i               ' 处置说明列位置
        End Select
    Next i
    
    ' 检查必要列是否存在，如果不存在则创建或报错
    If classificationCol = 0 Then
        ' 如果分类列不存在，在最后一列右侧创建新列
        classificationCol = lastCol + 1
        ws.Cells(1, classificationCol).Value = "分类"   ' 设置列标题
    End If
    
    ' 检查异常说明和步骤名称列是否都存在
    If exceptionCol = 0 Or stepNameCol = 0 Then
        MsgBox "请确保数据表包含'异常说明'和'步骤名称'列", vbExclamation
        Exit Sub  ' 如果缺少必要列，退出程序
    End If
    
    ' 获取数据总行数（从第2行开始是数据）
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    ' ==================== 主要分类处理逻辑 ====================
    ' 按优先级顺序处理每一行数据（优先级从高到低）
    For i = 2 To lastRow
        ' 只处理分类列为空的行（避免重复处理）
        If IsEmpty(ws.Cells(i, classificationCol).Value) Then
            exceptionText = CStr(ws.Cells(i, exceptionCol).Value)    ' 获取异常说明文本
            stepNameText = CStr(ws.Cells(i, stepNameCol).Value)      ' 获取步骤名称文本
            
            ' 1. 禁run
            If InStr(1, exceptionText, "禁run", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "禁 run", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "禁RUN", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "禁 RUN", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "设备EAP异常", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "换液卡控", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "EQP Constraint", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "Lot Abnormal End", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "禁run"
                GoTo NextRow  ' 跳转到下一行，确保优先级
            End If
            
            ' 2. OOS
            If InStr(1, exceptionText, "OOS", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "OOS"
                GoTo NextRow
            End If
            
            ' 3. OOC
            If InStr(1, exceptionText, "OOC", vbTextCompare) > 0 Or _
            InStr(1, exceptionText, "连续七点向上或向下趋势", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "OOC"
                GoTo NextRow
            End If
            
            ' 4. STH
           If InStr(1, exceptionText, "找点异常", vbTextCompare) > 0 Or _
              InStr(1, exceptionText, "STH", vbTextCompare) > 0 Or _
              InStr(1, exceptionText, "sth", vbTextCompare) > 0 Or _
              InStr(1, exceptionText, "Sth", vbTextCompare) > 0 Or _
              ((InStr(1, exceptionText, "没有菜单", vbTextCompare) > 0 Or _
              InStr(1, exceptionText, "recipe", vbTextCompare) > 0 Or _
              InStr(1, exceptionText, "Recipe", vbTextCompare) > 0) And _
              InStr(1, stepNameText, "厚度量测", vbTextCompare) > 0) Then
               ws.Cells(i, classificationCol).Value = "STH"
               GoTo NextRow
            End If
            
            ' 5. RMS
            If InStr(1, exceptionText, "RMS", vbTextCompare) > 0 Or _
               ((InStr(1, exceptionText, "Recipe", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "recipe", vbTextCompare) > 0) And _
               InStr(1, stepNameText, "刻蚀", vbTextCompare) > 0) Then
                ws.Cells(i, classificationCol).Value = "RMS"
                GoTo NextRow
            End If
            
            ' 6. SRS
            If InStr(1, exceptionText, "SRS流程结束", vbTextCompare) > 0 And _
               InStr(1, exceptionText, "Owner", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "SRS"
                GoTo NextRow
            End If
            
            ' 7. STR
            If InStr(1, exceptionText, "MSTR", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "STR", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "STR"
                GoTo NextRow
            End If
            
            ' 8. Qtime
            If InStr(1, exceptionText, "从", vbTextCompare) > 0 And _
               InStr(1, exceptionText, "到", vbTextCompare) > 0 And _
               InStr(1, exceptionText, "超时", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "Qtime"
                GoTo NextRow
            End If
            
            ' 9. 机台报警
            If InStr(1, exceptionText, "Run货过程中出现异常", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "报警", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "宕机", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "设备异常", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "机台报警"
                GoTo NextRow
            End If
            
            ' 10. PIRUN
            If InStr(1, exceptionText, "PIRUN", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "pi-run", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "Pirun", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "PIRUN"
                GoTo NextRow
            End If
            
            ' 11. 表界面
            If InStr(1, exceptionText, "脏污", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "撕金不净", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "糊胶", vbTextCompare) > 0 Or _
               (InStr(1, exceptionText, "目检", vbTextCompare) > 0 And _
                InStr(1, exceptionText, "异常", vbTextCompare) > 0) Or _
                (InStr(1, stepNameText, "量测", vbTextCompare) > 0 And _
                InStr(1, exceptionText, "图形不一致", vbTextCompare) > 0) Or _
               (InStr(1, stepNameText, "目检", vbTextCompare) > 0 And _
                InStr(1, exceptionText, "异常", vbTextCompare) > 0) Then
                ws.Cells(i, classificationCol).Value = "表界面"
                GoTo NextRow
            End If
            
            ' 12. 过账异常 - 数据过账和处理异常
            If InStr(1, exceptionText, "分批", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "合批", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "跳站", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "切帐", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "TZ", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "批次异常结束", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "Hold回，待蚀刻课工程师处理", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "check in", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "check out", vbTextCompare) > 0 Or _
               (InStr(1, exceptionText, "R:MFG姓名", vbTextCompare) > 0 And _
               InStr(1, exceptionText, "Lot Hold 待蚀刻课工程师处理", vbTextCompare) > 0) Or _
               (InStr(1, exceptionText, "R:MFG姓名", vbTextCompare) > 0 And _
               InStr(1, exceptionText, "无法正常进入SRS流程，等待蚀刻课工程师处理", vbTextCompare) > 0) Or _
               InStr(1, exceptionText, "待确认是否作业完成", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "待工程师跳至下一站点", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "过账异常"
                GoTo NextRow
            End If
            
            ' 13. 实验 - 实验相关异常
            If InStr(1, exceptionText, "实验片", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "报废", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "试验", vbTextCompare) > 0 Or _
               InStr(1, exceptionText, "实验", vbTextCompare) > 0 Then
                ws.Cells(i, classificationCol).Value = "实验"
                GoTo NextRow
            End If
            
            ' 14. 其他 - 以上都不匹配的归为其他
            ws.Cells(i, classificationCol).Value = "其他"
        End If
        
NextRow:  ' 跳转标签，用于跳过已处理的行
    Next i
    ' 自动调整分类列的列宽以适应内容
    ws.Columns(classificationCol).AutoFit
    
    ' 显示处理结果消息框
    ShowResultsInMessageBox ws, lastRow - 1, classificationCol
End Sub

' ================================================================
' 子程序：显示结果消息框
' 功能：弹出窗口显示分类处理结果摘要
' ================================================================
Sub ShowResultsInMessageBox(ws As Worksheet, totalRows As Long, classificationCol As Long)
    Dim resultMsg As String       ' 消息文本
    Dim categories As Variant     ' 分类类型数组
    Dim counts() As Long          ' 数量统计数组
    Dim i As Long                 ' 循环计数器
    
    ' 定义分类顺序
    categories = Array("禁run", "OOS", "OOC", "STH", "RMS", "SRS", "STR", "Qtime", "机台报警", "PIRUN", "表界面", "过账异常", "实验", "其他")
    ReDim counts(LBound(categories) To UBound(categories))  ' 重新定义数组大小
    
    ' 统计每个分类的数量
    For i = LBound(categories) To UBound(categories)
        counts(i) = Application.CountIf(ws.Columns(classificationCol), categories(i))
    Next i
    
    ' 构建消息文本内容
    resultMsg = "分类标记完成：" & vbNewLine & _
               "? 共处理 " & totalRows & " 行数据" & vbNewLine
               
    ' 添加每个分类的统计结果
    For i = LBound(categories) To UBound(categories)
        resultMsg = resultMsg & "? 标记" & categories(i) & " " & counts(i) & " 条" & vbNewLine
    Next i
    
    ' 显示消息框（带信息图标和确定按钮）
    MsgBox resultMsg, vbInformation + vbOKOnly, "分类结果"
End Sub
