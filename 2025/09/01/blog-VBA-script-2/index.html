VBA脚本：用于数据二次清洗

辅助脚本（by Uncle-yang）

借助AI写的一个小脚本，用于自动筛选异常类型并分类-层级：

Sub CreateOOSOOCLayerSummary()
    Dim ws As Worksheet, wsSummary As Worksheet
    Dim lastRow As Long, i As Long, j As Long, k As Long
    Dim dict As Object
    Dim classificationCol As Long, stepNameCol As Long
    Dim rng As Range
    
    ' 设置步骤名称分类（根据图片中的层）
    Dim steps As Variant
    steps = Array("AM", "CS", "GP", "GW", "MR", "MT", "SN", "TO", "AP", _
                 "N1", "NP", "OC", "PB", "PP", "SC", "FO", "M2", "M1", "JTE")
    
    ' 设置工作表对象
    Set ws = ThisWorkbook.ActiveSheet
    Set dict = CreateObject("Scripting.Dictionary")
    
    ' 查找关键列位置
    Dim lastCol As Long
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For i = 1 To lastCol
        Select Case ws.Cells(1, i).Value
            Case "分类": classificationCol = i
            Case "步骤名称": stepNameCol = i
        End Select
    Next i
    
    ' 检查必要列
    If classificationCol = 0 Or stepNameCol = 0 Then
        MsgBox "请确保数据表包含'分类'和'步骤名称'列", vbExclamation
        Exit Sub
    End If
    
    ' 获取数据行数
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    ' 创建或清空汇总表
    On Error Resume Next
    Set wsSummary = ThisWorkbook.Worksheets("OOS_OOC层统计")
    If wsSummary Is Nothing Then
        Set wsSummary = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        wsSummary.Name = "OOS_OOC层统计"
    Else
        wsSummary.Cells.Clear
    End If
    On Error GoTo 0
    
    ' 创建统计表格（按照图片中的两行布局）
    With wsSummary
        ' 第一组层（图片上半部分）
        .Range("A1").Value = "layer"
        For i = 0 To 8 ' AM到AP共9个
            .Cells(1, i + 2).Value = steps(i)
        Next i
        
        ' 第二组层（图片下半部分）
        .Range("A3").Value = "layer"
        For i = 9 To UBound(steps) ' N1到JTE共10个
            .Cells(3, i - 7).Value = steps(i) ' 调整列位置
        Next i
        
        ' 设置OOC/OOS标题
        .Range("A2").Value = "OOC/OOS"
        .Range("A4").Value = "OOC/OOS"
        
        ' 初始化计数为0
        For i = 0 To UBound(steps)
            dict.Add steps(i), 0
        Next i
        
        ' 统计OOS/OOC数量
        For i = 2 To lastRow
            If ws.Cells(i, classificationCol).Value = "OOS" Or ws.Cells(i, classificationCol).Value = "OOC" Then
                Dim stepName As String
                stepName = ws.Cells(i, stepNameCol).Value
                
                ' 检查步骤名称是否匹配
                For j = 0 To UBound(steps)
                    If InStr(1, stepName, steps(j), vbTextCompare) > 0 Then
                        dict(steps(j)) = dict(steps(j)) + 1
                        Exit For
                    End If
                Next j
            End If
        Next i
        
        ' 填充第一组数据（图片上半部分）
        For i = 0 To 8
            .Cells(2, i + 2).Value = dict(steps(i))
        Next i
        
        ' 填充第二组数据（图片下半部分）
        For i = 9 To UBound(steps)
            .Cells(4, i - 7).Value = dict(steps(i))
        Next i
        
        ' 设置表格格式
        .Range("A1").Resize(4, 10).Borders.LineStyle = xlContinuous
        .Range("A1:A4").Font.Bold = True
        .Range("A1:J1").Font.Bold = True
        .Range("A3:J3").Font.Bold = True
        
        ' 自动调整列宽
        .Columns.AutoFit
        
        ' 设置条件格式（三色阶）
        Dim dataRng As Range
        Set dataRng = Union(.Range("B2:J2"), .Range("B4:J4"))
        
        dataRng.FormatConditions.Delete
        With dataRng.FormatConditions.AddColorScale(ColorScaleType:=3)
            ' 最小值 - 黄色
            .ColorScaleCriteria(1).Type = xlConditionValueLowestValue
            .ColorScaleCriteria(1).FormatColor.Color = RGB(255, 255, 0)
            
            ' 中间值 - 绿色
            .ColorScaleCriteria(2).Type = xlConditionValuePercentile
            .ColorScaleCriteria(2).Value = 50
            .ColorScaleCriteria(2).FormatColor.Color = RGB(146, 208, 80)
            
            ' 最大值 - 粉红色
            .ColorScaleCriteria(3).Type = xlConditionValueHighestValue
            .ColorScaleCriteria(3).FormatColor.Color = RGB(255, 192, 203)
        End With
        
        ' 设置蓝紫色边框（RGB(112, 48, 160)）
        With .Range("A1:J4").Borders
            .LineStyle = xlContinuous
            .Color = RGB(112, 48, 160)
            .Weight = xlThin
        End With
    End With
    
    MsgBox "OOS/OOC层统计表已创建完成！", vbInformation, "完成"
End Sub
